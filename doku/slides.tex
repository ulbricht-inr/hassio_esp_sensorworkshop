\section{Motivation}

\begin{frame}{Was haben wir heute vor~\ldots}
  \begin{itemize}
    \item Wir bauen ein \textit{Smart-Home-System} mit einer Steuerzentrale,
      einem Sensor und einem Aktor.
    \item Wir benutzen dafür nur Komponenten, die quelloffen sind und bei denen
      wir verstehen können, was im Hintergrund passiert (also nichts von Apple
      oder Microsoft).
  \end{itemize}

  \begin{block}{Zielstellung}
    Sie bekommen heute eine ausführliche Anleitung zum Zusammenbau eines kleinen
    Smart-Home-Systems, das Sie auch leicht zu Hause nachbauen können. Alle
    notwendigen Schritte können mit diesem PDF und dem Repository unter
    \url{https://github.com/ulbricht-inr/hassio_esp_sensorworkshop}
    nachvollzogen werden.
  \end{block}
\end{frame}

\section{Hardware}

\begin{frame}{Hardware, die wir benutzen I}
  \begin{block}{Raspberry Pi}
    \begin{itemize}
      \item Mini-Computer mit mehreren Kernen und genug Rechenleistung, um als
        Office-PC genutzt zu werden. Als Betriebssystem kommt Linux zum Einsatz.
      \item Gebaut als Lehr- und Lernplattform, mit Zugang zu
        Hardware-Komponenten wie GPIO, ADC und Feldbussen.
      \item Besitzt eine sehr große Online-Community mit vielen Anleitungen und
        Hilfestellungen. \textbf{Jeder Schüler sollte einen haben!}
    \end{itemize}
    \centering
    \includegraphics[width=0.25\textwidth]{bilder/rpi.png}
  \end{block}
\end{frame}

\section{Home Assistant}

\begin{frame}{Home Assistant öffnen}
  \begin{block}{1. Am PC anmelden}
    \begin{center}
      \begin{tabular}{rl}
        \textbf{Benutzername:} & \texttt{famuly} \\
        \textbf{Passwort:}     & steht an der Tafel
      \end{tabular}
    \end{center}
  \end{block}

  \begin{block}{2. Home Assistant öffnen und anmelden}
    Öffne Firefox, rufe die Home-Assistant-Adresse von der Tafel auf und nutze:
    \begin{center}
      \begin{tabular}{rl}
        \textbf{Benutzername:} & \texttt{student} \\
        \textbf{Passwort:}     & \texttt{htwk2026}
      \end{tabular}
    \end{center}
  \end{block}
\end{frame}

\begin{frame}{Mit deinem Home Assistant verbinden}
  \begin{enumerate}
    \item Ermittle die IP-Adresse deines Raspberry Pi und schreibe sie dir auf.
    \item Öffne auf deinem PC einen Webbrowser.
    \item Rufe die folgende Adresse auf:

      \begin{center}
        \Large\texttt{http://<IP-Adresse>:8123}
      \end{center}

    \item Erstelle einen neuen Benutzer.
  \end{enumerate}
\end{frame}

\begin{frame}{ESPHome-Datei öffnen}
  \begin{enumerate}
    \item Öffne auf dem Desktop den Ordner
      \texttt{hassio\_esp\_sensorworkshop}.
    \item Öffne darin den Unterordner \texttt{src}.
    \item Öffne die Textdatei \texttt{esphome\_start.yml}.
  \end{enumerate}

  \vspace{0.8em}

  \begin{block}{WLAN-Zugangsdaten}
    SSID und WLAN-Passwort gehören in die Datei
    \texttt{src/secrets.yaml}. Die ESPHome-Konfiguration liest sie von
    dort automatisch ein.
  \end{block}
\end{frame}

\begin{frame}{Code und Schaltung verbinden}
  \begin{block}{Deine Aufgabe}
    Schau dir die Datei \texttt{esphome\_start.yml} an. Versuche zu
    verstehen, wie der Code mit dem Schaltplan auf der vorherigen Folie
    zusammenhängt.
  \end{block}

  \vspace{0.6em}

  \begin{itemize}
    \item Ändere in der YML-Datei den Namen des Projekts in einen eigenen Namen.
    \item Wo findest du den ESP32 und den DHT11 im Code wieder?
    \item Welche Codezeile gehört zur Datenleitung an \texttt{GPIO4}?
    \item Welche beiden Messwerte liefert der Sensor?
    \item Welche Leitungen aus dem Schaltplan tauchen nicht im Code auf?
  \end{itemize}
\end{frame}

\begin{frame}{ESPHome kompilieren}
  \begin{enumerate}
    \item Öffne \texttt{secrets.yaml} und trage die WLAN-Daten von der
      Tafel ein.
    \item Verbinde den ESP32 mit einem USB-Kabel über den USB-Hub mit
      deinem Rechner.
    \item Öffne ein Terminal und wechsle in den Quellordner:
      \begin{center}
        \texttt{cd Desktop/hassio\_esp\_sensorworkshop/src}
      \end{center}
      {\footnotesize Nutze die \keystroke{Tab}-Taste zur Autovervollständigung.}
    \item Kompiliere die Konfiguration:
      \begin{center}
        \texttt{esphome compile esphome\_start.yml}
      \end{center}
    \item Übertrage den Code und starte den ESP32:
      \begin{center}
        \texttt{esphome run esphome\_start.yml}
      \end{center}
      {\footnotesize Du solltest jetzt sehen, dass der ESP32 den Code
        ausführt und sich mit dem WLAN verbindet.}
    \item Merke dir die IP-Adresse, die dein Sensor bekommt, und schreibe
      sie auf. Du findest sie im Log; sie sieht etwa so aus:
      \texttt{192.168.2.xxx}.
  \end{enumerate}
\end{frame}

\begin{frame}{Sensor in Home Assistant hinzufügen}
  \begin{enumerate}
    \item Wechsle im Webbrowser zurück zu Home Assistant.
    \item Öffne \textbf{Einstellungen $\rightarrow$ Geräte} und füge dort
      \textbf{ESPHome} hinzu.
    \item Trage als Hostnamen die IP-Adresse deines Sensors ein, die du dir
      auf der vorherigen Folie notiert hast.
  \end{enumerate}

  \vspace{1em}

  \begin{block}{Hostnamen eintragen}
    Beispiel: \texttt{192.168.2.\ldots}
  \end{block}
\end{frame}

\begin{frame}{Messwerte zum Dashboard hinzufügen}
  Du solltest deinen neuen Sensor jetzt in Home Assistant sehen.

  \vspace{1em}

  \begin{block}{Deine Aufgabe}
    Füge die beiden Messwerte deines Sensors zu deinem Dashboard hinzu:
    \begin{itemize}
      \item Temperatur
      \item Luftfeuchtigkeit
    \end{itemize}
  \end{block}
\end{frame}
